                },

                    SwaggerUIBundle.presets.apis,

                    SwaggerUIStandalonePreset

                docExpansion: "none",

                defaultModelsExpandDepth: -1,

                defaultModelExpandDepth: 2,

                displayRequestDuration: true,

                displayOperationId: false,

                showExtensions: false,

                persistAuthorization: false

            });

             * =====================================================
             * PERSONALIZAÇÕES APÓS O CARREGAMENTO DO SWAGGER
             * =====================================================
             */

            const aplicarPersonalizacoes = function() {


                /*
                 * Remove Schemas
                 */

                document
                    .querySelectorAll('.models')
                    .forEach(function(el) {

                        el.style.display = 'none';

                    });


                document
                    .querySelectorAll('.models-control')
                    .forEach(function(el) {

                        el.style.display = 'none';

                    });


                /*
                 * Remove Servers
                 */

                document
                    .querySelectorAll('.scheme-container')
                    .forEach(function(el) {

                        el.style.display = 'none';

                    });


                /*
                 * Remove Authorize
                 */

                document
                    .querySelectorAll('.auth-wrapper')
                    .forEach(function(el) {

                        el.style.display = 'none';

                    });


                /*
                 * Remove Download URL
                 */

                document
                    .querySelectorAll('.download-url-wrapper')
                    .forEach(function(el) {

                        el.style.display = 'none';

                    });

            };


            setTimeout(
                aplicarPersonalizacoes,
                500
            );


            document.addEventListener(
                'click',
                function() {

                    setTimeout(
                        aplicarPersonalizacoes,
                        100
                    );

                }
            );


        };
