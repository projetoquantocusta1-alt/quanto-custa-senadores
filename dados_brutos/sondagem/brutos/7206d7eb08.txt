    // Função para enviar dados ao servidor via Fetch (Upsert)
    function sincronizarResposta(perguntaId, dadosExtras = {}) {
        const token = document.querySelector('input[name="_token"]').value;
        const radioMarcado = document.querySelector(`input[name="respostas[${perguntaId}]"]:checked`);
        const opcaoId = radioMarcado ? radioMarcado.value : null;

        const payload = {
            pergunta_id: perguntaId
            , opcao_id: opcaoId
            , ...dadosExtras
        };

        fetch("https://transparencia.al.ce.gov.br/pesquisa/upsert", {
                method: 'POST'
                , headers: {
                    'Content-Type': 'application/json'
                    , 'X-CSRF-TOKEN': token
                    , 'Accept': 'application/json'
                }
                , body: JSON.stringify(payload)
            })
            .then(response => response.json())
            .then(data => console.log(`Pergunta ${perguntaId} sincronizada.`, data))
            .catch(error => console.error('Erro na sincronização:', error));
    }

    function salvarResposta(perguntaId, opcaoId) {
        sincronizarResposta(perguntaId, {
            opcao_id: opcaoId
        });
    }

    function salvarComentario(perguntaId) {
        const textarea = document.getElementById(`text_${perguntaId}`);
        if (!textarea) return;
        sincronizarResposta(perguntaId, {
            comentario: textarea.value
        });
    }

    document.addEventListener('DOMContentLoaded', function() {
        // Event listener para mudança de opções (Radio)
        document.querySelectorAll('.radio-opcao').forEach(radio => {
            radio.addEventListener('change', function() {
                const pId = this.getAttribute('data-pergunta');
                salvarResposta(pId, this.value);

                const extraDiv = document.getElementById(`extra_${pId}`);
                if (extraDiv) {
                    if (this.getAttribute('data-show-extra') === 'true') {
                        extraDiv.classList.remove('d-none');
                        extraDiv.querySelector('textarea').focus();
                    } else {
                        extraDiv.classList.add('d-none');
                        extraDiv.querySelector('textarea').value = '';
                    }
                }
            });
        });

        // Validação e Envio do Formulário
        document.getElementById('formPesquisa').addEventListener('submit', function(e) {
            e.preventDefault();

            // 1. Validar Obrigatoriedade (1 a 6)
            for (let i = 1; i <= 6; i++) {
                const radioMarcado = document.querySelector(`input[name="respostas[${i}]"]:checked`);

                if (!radioMarcado) {
                    Swal.fire({
                        icon: 'error'
                        , title: 'Questão Obrigatória'
                        , text: `A pergunta ${i} precisa ser respondida.`
                        , confirmButtonColor: '#2e7d32'
                    });

                    document.getElementById(`card_pergunta_${i}`).scrollIntoView({
                        behavior: 'smooth'
                        , block: 'center'
                    });
                    return;
                }
            }

            // 2. Validação Específica do "Não Encontrei" na Pergunta 4
            const p4Radio = document.querySelector(`input[name="respostas[4]"]:checked`);
            if (p4Radio) {
                const textoOpcao4 = p4Radio.getAttribute('data-texto');
                const textarea4 = document.getElementById('text_4');

                if (textoOpcao4.includes('não encontrei') && textarea4.value.trim() === '') {
                    Swal.fire({
                        icon: 'warning'
                        , title: 'Falta um detalhe'
                        , text: 'Por favor, descreva o que você não encontrou na questão 4.'
                        , confirmButtonColor: '#2e7d32'
                    });
                    textarea4.focus();
                    return;
                }
            }

            // 3. Finalização
            Swal.fire({
                icon: 'success'
                , title: 'Sucesso!'
                , text: 'Suas respostas foram salvas.'
                , showConfirmButton: false
                , timer: 1500
            }).then(() => {
                window.location.href = "https://transparencia.al.ce.gov.br/grafico-pesquisa";
            });
        });
    });

    <script type="module">
        const root = $(":root")
        const defaultPSize = parseInt(root.css('font-size'))
        const defaultHSize = parseInt($('h1, h2, h3, h4, h5, h6').css('font-size'))

        function setFontSize(size) {
            $('p').each(function() {
                $(this).css('font-size', defaultPSize + 8 * size + 'px');
            })

            $('a').each(function() {
                $(this).css('font-size', defaultPSize + 8 * size + 'px');
            })

            $('h1, h2, h3, h4, h5, h6').each(function() {
                $(this).css('font-size', defaultHSize + 8 * size + 'px');
            })

            localStorage.setItem("font-size", size)
        }

        function increaseFontSize() {
            const currentSize = parseInt(localStorage.getItem("font-size"))
            if (currentSize < 2) {
                setFontSize(currentSize + 1)
            }
        }

        function decreaseFontSize() {
            const currentSize = parseInt(localStorage.getItem("font-size"))
            if (currentSize > 0) {
                setFontSize(currentSize - 1)
            }
        }

            if (localStorage.getItem("high-contrast") == 'true') {
                root.addClass("high-contrast")

            if (!localStorage.getItem("font-size")) {
