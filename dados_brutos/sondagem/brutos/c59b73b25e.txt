    function trocarSelo(ano) {
        const selo = document.getElementById('selo-diamante');
        const botoes = document.querySelectorAll('.btn-ano');
        
        selo.classList.add('fade-out');
        
        setTimeout(() => {
            if (ano === 2026) {
                selo.src = "https://transparencia.assembleia.pr.leg.br/assets/img/diamante-2026.png";
                selo.alt = "Selo Prêmio Diamante de Transparência 2026";
                selo.title = "Selo do Prêmio Diamante de Transparência 2026";
            } else if (ano === 2025) {
                selo.src = "https://transparencia.assembleia.pr.leg.br/assets/img/diamante-2025.png";
                selo.alt = "Selo Prêmio Diamante de Transparência 2025";
                selo.title = "Selo do Prêmio Diamante de Transparência 2025";
            } else if (ano === 2024) {
                selo.src = "https://transparencia.assembleia.pr.leg.br/assets/img/diamante.png";
                selo.alt = "Selo Prêmio Diamante de Transparência 2024";
                selo.title = "Selo do Prêmio Diamante de Transparência 2024";
            }
            
            selo.classList.remove('fade-out');
            selo.classList.add('fade-in');
            
            setTimeout(() => {
                selo.classList.remove('fade-in');
            }, 600);
        }, 300);
        
        botoes.forEach(btn => {
            if (parseInt(btn.dataset.ano) === ano) {
                btn.classList.add('active');
            } else {
                btn.classList.remove('active');
            }
        });
    }
